# Payroll Management System
# Without Classes and Objects

employees = []


# Add Employee
def add_employee():
    employee_id = int(input("Enter Employee ID: "))
    name = input("Enter Employee Name: ")
    basic_salary = float(input("Enter Basic Salary: "))

    hra = basic_salary * 0.20
    da = basic_salary * 0.10
    tax = basic_salary * 0.05

    employee = [
        employee_id,
        name,
        basic_salary,
        hra,
        da,
        tax
    ]

    employees.append(employee)

    print("Employee added successfully!")


# Display All Employees
def display_employees():

    if len(employees) == 0:
        print("No employees found.")
        return

    print("\n===== EMPLOYEE DETAILS =====")

    for employee in employees:

        employee_id = employee[0]
        name = employee[1]
        basic_salary = employee[2]
        hra = employee[3]
        da = employee[4]
        tax = employee[5]

        gross_salary = basic_salary + hra + da
        net_salary = gross_salary - tax

        print("-----------------------------------")
        print("Employee ID   :", employee_id)
        print("Employee Name :", name)
        print("Basic Salary  :", basic_salary)
        print("HRA           :", hra)
        print("DA            :", da)
        print("Gross Salary  :", gross_salary)
        print("Tax           :", tax)
        print("Net Salary    :", net_salary)
        print("-----------------------------------")


# Search Employee
def search_employee():

    employee_id = int(input("Enter Employee ID to search: "))

    for employee in employees:

        if employee[0] == employee_id:

            basic_salary = employee[2]
            hra = employee[3]
            da = employee[4]
            tax = employee[5]

            gross_salary = basic_salary + hra + da
            net_salary = gross_salary - tax

            print("\nEmployee Found:")
            print("-----------------------------------")
            print("Employee ID   :", employee[0])
            print("Employee Name :", employee[1])
            print("Basic Salary  :", basic_salary)
            print("HRA           :", hra)
            print("DA            :", da)
            print("Gross Salary  :", gross_salary)
            print("Tax           :", tax)
            print("Net Salary    :", net_salary)
            print("-----------------------------------")

            return

    print("Employee not found.")


# Generate Payroll
def generate_payroll():

    employee_id = int(input("Enter Employee ID: "))

    for employee in employees:

        if employee[0] == employee_id:

            basic_salary = employee[2]
            hra = employee[3]
            da = employee[4]
            tax = employee[5]

            gross_salary = basic_salary + hra + da
            net_salary = gross_salary - tax

            print("\n========== PAYSLIP ==========")
            print("Employee ID   :", employee[0])
            print("Employee Name :", employee[1])
            print("------------------------------")
            print(f"Basic Salary  : {basic_salary:.2f}")
            print(f"HRA (20%)     : {hra:.2f}")
            print(f"DA (10%)      : {da:.2f}")
            print(f"Gross Salary  : {gross_salary:.2f}")
            print(f"Tax (5%)      : {tax:.2f}")
            print("------------------------------")
            print(f"Net Salary    : {net_salary:.2f}")
            print("==============================")

            return

    print("Employee not found.")


# Main Program
while True:

    print("\n===== PAYROLL MANAGEMENT SYSTEM =====")
    print("1. Add Employee")
    print("2. Display Employees")
    print("3. Search Employee")
    print("4. Generate Payroll")
    print("5. Exit")

    choice = int(input("Enter your choice: "))

    if choice == 1:
        add_employee()

    elif choice == 2:
        display_employees()

    elif choice == 3:
        search_employee()

    elif choice == 4:
        generate_payroll()

    elif choice == 5:
        print("Thank you for using Payroll Management System!")
        break

    else:
        print("Invalid choice! Please try again.")
